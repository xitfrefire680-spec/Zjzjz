local SPOOF_KEY, SPOOF_VERSION = "__SenaV4StealAnEggMovementSpoof", 436
local SPOOF_FIELDS = {
    "LastObservedSample","LastSample","LastGoodSample","LastGameplayTrustedSample",
    "LastValidatedSample","LastValidatedGroundedSample","LastConfirmedGroundSample",

local HEADROOM, MIN_SPEED, MAX_SPEED = 1.35, 200, 1000 * 1.35

local function GetSpoof()
    local g = GetGenv()
    local s = g[SPOOF_KEY]
    if type(s) ~= "table" or s.Version ~= SPOOF_VERSION then
        if type(s) == "table" then s.Enabled = false end
        s = { Version = SPOOF_VERSION, Installed = false, Enabled = false,
              Mode = "StateBaseline", Humanoid = nil, RootPart = nil, WalkSpeed = 600 }

end

local HoverPad, HoverOwnerId, HoverConns = { part = nil }, tostring(math.random(100000, 999999)), {}
local function SyncHoverPad()
    local char = LP.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum  = char and char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end
    if not HoverPad.part or not HoverPad.part.Parent then
        for _, c in ipairs(workspace:GetChildren()) do
            if c.Name == "__SenaV4HoverPad" and c ~= HoverPad.part then
                pcall(function() c:Destroy() end)
            end
        end
        local p = Instance.new("Part")
        p.Name = "__SenaV4HoverPad"
        p:SetAttribute("SenaV4RuntimeOwner", HoverOwnerId)
        p.Anchored = true p.CanCollide = true p.CanQuery = true
        p.CanTouch = false p.CastShadow = false p.Transparency = 1
        p.Material = Enum.Material.SmoothPlastic
        p.Size = Vector3.new(8, 1, 8)
        HoverPad.part = p p.Parent = workspace
    end
    local yOff = hum.RigType == Enum.HumanoidRigType.R15
        and (root.Size.Y * 0.5 + math.max(hum.HipHeight, 0.5))
        or  (root.Size.Y * 0.5 + 2)
    HoverPad.part.CFrame = CFrame.new(root.Position.X, root.Position.Y - yOff - 0.55, root.Position.Z)
end
local function StartHoverPad()

    if not HoverConns[1] then HoverConns[1] = RunService.PreSimulation:Connect(SyncHoverPad) end
    if not HoverConns[2] then HoverConns[2] = RunService.PostSimulation:Connect(SyncHoverPad) end
end

local function FindIntegrityState()
    if type(getgc) ~= "function" then return nil end
    local ok, list = pcall(getgc, true)
    if not ok or type(list) ~= "table" then return nil end
    for _, obj in ipairs(list) do
        if type(obj) == "table" then
            local hit = false
            pcall(function()
                hit = rawget(obj, "Player") == LP
                    and rawget(obj, "Character") == LP.Character
                    and type(rawget(obj, "SampleHistory")) == "table"
                    and type(rawget(obj, "Evidence")) == "table"
            end)
            if hit then return obj end
        end
    end
    return nil
end

local function SpoofState(state, walkSpeed)
    if type(state) ~= "table" then return end
    local root = state.RootPart
    local cf = root and root.Parent and root.CFrame
    local pos = cf and cf.Position
    local function apply(t)
        if type(t) == "table" then
            rawset(t, "WalkSpeed", walkSpeed)
            if cf then
                rawset(t, "CFrame", cf) rawset(t, "Position", pos)
                rawset(t, "LinearVelocity", Vector3.zero)
                rawset(t, "AngularVelocity", Vector3.zero)
            end
        end
    end
    for _, f in ipairs(SPOOF_FIELDS) do apply(rawget(state, f)) end
    for _, v in pairs(rawget(state, "SampleHistory") or {}) do apply(v) end
    for _, v in pairs(rawget(state, "SafeGroundCheckpoints") or {}) do apply(v) end
    local ev = rawget(state, "Evidence")
    if type(ev) == "table" then ev.Speed=0 ev.Teleport=0 ev.Flight=0 end
    local imp = rawget(state, "ImpulseContext")
    if type(imp) == "table" then
        local mhs  = math.max(tonumber(imp.MaxHorizontalSpeed) or 0, walkSpeed)
        local span = math.max((tonumber(imp.ExpiresAt) or 0) - (tonumber(imp.StartedAt) or 0), 0.15)
        imp.MaxHorizontalSpeed = mhs
        if pos then imp.OriginPosition = pos end
        imp.MaxHorizontalDistance = math.max(tonumber(imp.MaxHorizontalDistance) or 0, mhs * span + 48)
        apply(imp.PreMovementSafeSample)
    end
    if rawget(state, "CorrectionContext") == nil then
        rawset(state, "FirstSuspiciousAt", nil)
        if rawget(state, "ThreatLevel") == "Observing" then rawset(state, "ThreatLevel", "Trusted") end
    end
end

local function getCarriedEggTool()
    local char = LP.Character
    if not char then return nil end
    for _, t in ipairs(char:GetChildren()) do
        if t:IsA("Tool") then
            local isEgg = false
            if EggToolDisplay and EggToolDisplay.IsEggTool then
                local ok, r = pcall(EggToolDisplay.IsEggTool, t)
                if ok and r then isEgg = true end
            end
            if not isEgg and tostring(t.Name):lower():find("egg") then isEgg = true end
            if isEgg then return t end
        end
    end
    return nil
end

local function getToolUid(t)
    if not t then return nil end
    local uid = t:GetAttribute("EggUid") or t:GetAttribute("Uid")
    if uid then return uid end
    if EggToolDisplay and typeof(EggToolDisplay.GetToolUid) == "function" then
        local ok, r = pcall(EggToolDisplay.GetToolUid, t)
        if ok and r then return r end
    end
    return nil
end

local function findCarriedUid()
    local t = getCarriedEggTool()
    if t then
        local uid = getToolUid(t)
        if uid then return uid end
    end
    if EggState and EggState.ReadFieldEggs then
        local ok, snap = pcall(EggState.ReadFieldEggs)
        if ok and snap and snap.Records then
            for _, r in ipairs(snap.Records) do
                if r.State == "Carried" and r.CarrierUserId == LP.UserId then
                    return r.Uid
                end
            end
        end
    end
    return nil
end

local function isCarryingEgg()
    if getCarriedEggTool() then return true, findCarriedUid() end
    if EggState and EggState.ReadFieldEggs then
        local ok, snap = pcall(EggState.ReadFieldEggs)
        if ok and snap and snap.Records then
            for _, r in ipairs(snap.Records) do
                if r.State == "Carried" and r.CarrierUserId == LP.UserId then
                    return true, r.Uid
                end
            end
        end
    end
    return false
end

local function getEggInfo(record)
    if not record then return { name="?", rarity="?", icon=nil, rank=0 } end
    local cat = record.AssetCategory or record.Category or "?"
    local dir = AssetsData and (AssetsData.Directory or AssetsData)
    local info = dir and dir[cat]
    if type(info) ~= "table" then return { name=cat, rarity="?", icon=nil, rank=0 } end
    local rarity = info.Rarity
    local rname = (type(rarity) == "table" and (rarity.DisplayName or rarity._id)) or "?"
    local display = info.DisplayName or info._id or cat
    local icon = nil
    for _, key in ipairs({"Icon","Image","Thumbnail","IconId","ImageId","Texture"}) do
        local v = info[key] or (info.Egg and info.Egg[key])
        if type(v) == "string" and v ~= "" then
            icon = v:match("^%d+$") and ("rbxassetid://"..v) or v break
        elseif type(v) == "number" then icon = "rbxassetid://"..v break end
    end
    return { name=display, rarity=rname, icon=icon, rank=RARITY_RANK[rname] or 0, color=rarity and rarity.Color or nil }
end

-- 1) VOO CLASSICO — BodyVelocity
function Fly.classic(targetPos, maxTime)
    if typeof(targetPos) ~= "Vector3" then return false end
    local hum, hrp = getChar()
    if not hum or not hrp then return false end
    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(1e6, 1e6, 1e6)
    bv.P = 5000
    bv.Parent = hrp
    local t0 = tick()
    maxTime = maxTime or 15
    while not HUB.dead do
        if tick() - t0 > maxTime then break end
        local _, r = getChar()
        if not r then break end
        local to = targetPos - r.Position
        local dist = to.Magnitude
        if dist < 2 then break end
        local speed = HUB.flySpeed
        if dist < 15 then speed = math.max(30, (dist/15) * HUB.flySpeed) end
        bv.Velocity = to.Unit * speed
        tickSpoof()
        RunService.Heartbeat:Wait()
    end
    bv:Destroy()
    return true
end

-- 2) VOO TWEEN PRO — takeoff -> cruise -> descend
function Fly.tween(targetPos, maxTime, useCruise, zigzag)
    if typeof(targetPos) ~= "Vector3" then return false end
    local hum, hrp = getChar()
    if not hum or not hrp then return false end

    local origAutoRotate = hum.AutoRotate
    pcall(function() hum.AutoRotate = false end)

    local startPos = hrp.Position
    local startDist = (targetPos - startPos).Magnitude
    local cruiseY = useCruise and HUB.flyCruiseY or nil
    if cruiseY and startDist < 60 then cruiseY = nil end

    local phase = cruiseY and "takeoff" or "cruise"
    local zigT = 0
    local zigLast = 0
    local t0 = tick()
    maxTime = maxTime or (startDist / HUB.flySpeed * 2 + 10)

    while not HUB.dead do
        if tick() - t0 > maxTime then break end
        local _, r = getChar()
        if not r then break end

        local pos = r.Position
        local targetPos2 = targetPos

        if cruiseY then
            if phase == "takeoff" then
                if pos.Y >= cruiseY - 0.5 then
                    phase = "cruise"
                else
                    targetPos2 = Vector3.new(pos.X, cruiseY, pos.Z)
                end
            end
            if phase == "cruise" then
                local hDelta = Vector3.new(targetPos.X - pos.X, 0, targetPos.Z - pos.Z).Magnitude
                if hDelta <= 4 then
                    phase = "descend"
                else
                    targetPos2 = Vector3.new(targetPos.X, cruiseY, targetPos.Z)
                end
            end
            if phase == "descend" then
                targetPos2 = targetPos
            end
        end

        if zigzag then
            local to = targetPos2 - pos
            local horiz = Vector3.new(to.X, 0, to.Z)
            local hMag = horiz.Magnitude
            if hMag > 12 then
                local unit = horiz.Unit
                local perp = Vector3.new(-unit.Z, 0, unit.X)
                local width = math.min(HUB.flyZigzagWidth, hMag * 0.3)
                if hMag < 120 then width = width * (hMag / 120) end
                zigT = zigT + (HUB.flySpeed * 0.016)
                local zig = width * math.sin(zigT / 900 * math.pi * 2)
                local delta = zig - zigLast
                zigLast = zig
                targetPos2 = targetPos2 + perp * delta
            else
                zigLast = 0
            end
        end

        local to = targetPos2 - pos
        local dist = to.Magnitude
        if dist < 1.5 then
            if phase == "descend" or not cruiseY then break end
        end

        local step = math.min(HUB.flySpeed * RunService.Heartbeat:Wait(), dist)
        if step <= 0 then step = 0.5 end
        local newPos = pos + to.Unit * step

        pcall(function()
            r.CFrame = CFrame.new(newPos, newPos + to.Unit)
            r.AssemblyLinearVelocity = Vector3.zero
            r.AssemblyAngularVelocity = Vector3.zero
        end)
        tickSpoof()
    end

    pcall(function() hum.AutoRotate = origAutoRotate end)
    return true
end

-- 3) VOO AEREO — hover em Y alto + anim
local function stopAnimations()
    pcall(function()
        local hum = getHum()
        if not hum then return end
        local animator = hum:FindFirstChildWhichIsA("Animator")
        if animator then
            for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
                track:Stop(0)
            end
        end
    end)
end
local function setAnimateDisabled(v)
    local char = LP.Character
    if not char then return end
    local anim = char:FindFirstChild("Animate")
    if anim then pcall(function() anim.Disabled = v end) end
end

function Fly.aerial(targetPos, maxTime)
    if typeof(targetPos) ~= "Vector3" then return false end
    local hum, hrp = getChar()
    if not hum or not hrp then return false end

    local origAutoRotate = hum.AutoRotate
    pcall(function() hum.AutoRotate = true end)

    setAnimateDisabled(true)
    stopAnimations()

    local hoverY = HUB.flyCruiseY
    pcall(function()
        hrp.CFrame = CFrame.new(hrp.Position.X, hoverY, hrp.Position.Z) * hrp.CFrame.Rotation
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end)
    SyncHoverPad()

    local t0 = tick()
    local startDist = (targetPos - hrp.Position).Magnitude
    maxTime = maxTime or (startDist / HUB.flySpeed * 2 + 12)

    while not HUB.dead do
        if tick() - t0 > maxTime then break end
        local _, r = getChar()
        if not r then break end

        local pos = r.Position
        local targetPos2 = Vector3.new(targetPos.X, hoverY, targetPos.Z)
        local to = targetPos2 - pos
        local dist = to.Magnitude
        if dist < 4 then
            targetPos2 = targetPos
            to = targetPos2 - pos
            dist = to.Magnitude
            if dist < 2 then break end
        end

        local step = math.min(HUB.flySpeed * RunService.Heartbeat:Wait(), dist)
        if step <= 0 then step = 0.5 end
        local newPos = pos + to.Unit * step
        pcall(function()
            r.CFrame = CFrame.new(newPos, newPos + to.Unit)
            r.AssemblyLinearVelocity = Vector3.zero
            r.AssemblyAngularVelocity = Vector3.zero
        end)
        tickSpoof()
    end

    setAnimateDisabled(false)
    pcall(function() hum.AutoRotate = origAutoRotate end)
    return true
end

-- 4) VOO ZIGZAG

-- 5) VOO INSTANT — glide reto rapido
function Fly.instant(targetPos, maxTime)
    if typeof(targetPos) ~= "Vector3" then return false end
    local _, hrp = getChar()
    if not hrp then return false end

    local t0 = tick()
    maxTime = maxTime or 12

    while not HUB.dead do
        if tick() - t0 > maxTime then break end
        local _, r = getChar()
        if not r then break end
        local to = targetPos - r.Position
        local dist = to.Magnitude
        if dist < 2 then break end
        local step = math.min(HUB.flySpeed * RunService.Heartbeat:Wait() * 3, dist)
        pcall(function()
            r.CFrame = CFrame.new(r.Position + to.Unit * step)
            r.AssemblyLinearVelocity = Vector3.zero
            r.AssemblyAngularVelocity = Vector3.zero
        end)
        tickSpoof()
    end
    return true
end

-- DISPATCHER
function Fly.go(targetPos, maxTime)
    local mode = HUB.flyMode or "Tween Pro"
    if mode == "Classico"   then return Fly.classic(targetPos, maxTime) end
    if mode == "Tween Pro"  then return Fly.tween(targetPos, maxTime, true, false) end
    if mode == "Aereo"      then return Fly.aerial(targetPos, maxTime) end
    if mode == "ZigZag"     then return Fly.zigzag(targetPos, maxTime) end
    if mode == "Instant"    then return Fly.instant(targetPos, maxTime) end
    return Fly.classic(targetPos, maxTime)
end

local function flyTo(targetPos, maxTime) return Fly.go(targetPos, maxTime) end

local ReturnState = { uid = nil, slotKey = nil }

local function doDropRemote()

local function doCarryRemote(uid, slotKey)
    local args = { Uid = uid, FirstAreaSlotKey = slotKey }
    if EggWorld and EggWorld.AskFieldEggCarry then
        pcall(function() EggWorld.AskFieldEggCarry:InvokeServer(args) end)
    end
    if EggState and EggState.CarryFieldEgg then
        pcall(function() EggState.CarryFieldEgg(uid, slotKey) end)
    end
end

local function captureReturnContext()
    ReturnState.uid = nil
    ReturnState.slotKey = nil
    local uid = findCarriedUid()
    ReturnState.uid = uid
    if uid and AreaEggSlotId and AreaEggSlotId.LooksLikeFirstAreaUid
       and AreaEggSlotId.LooksLikeFirstAreaUid(uid) then
        local ok, rec = pcall(function()
            return EggState and EggState.ReadFieldEgg and EggState.ReadFieldEgg(uid)
        end)
        if ok and rec then
            ReturnState.slotKey = AreaEggSlotId.SlotKey(rec.AreaId, rec.NestId)
        end
    end
    return uid ~= nil
end

local function dropPickIsland()
    if not ReturnState.uid then
        ReturnState.uid = findCarriedUid()
    end
    if not ReturnState.uid then return false end

    if not ReturnState.slotKey and AreaEggSlotId and AreaEggSlotId.LooksLikeFirstAreaUid
       and AreaEggSlotId.LooksLikeFirstAreaUid(ReturnState.uid) then
        local ok, rec = pcall(function()
            return EggState and EggState.ReadFieldEgg and EggState.ReadFieldEgg(ReturnState.uid)
        end)
        if ok and rec then
            ReturnState.slotKey = AreaEggSlotId.SlotKey(rec.AreaId, rec.NestId)
        end
    end

    doDropRemote()

    local t0 = tick()
    while tick() - t0 < 0.4 do
        if not getCarriedEggTool() then break end
        task.wait(0.02)
    end

    t0 = tick()
    local ok = false
    while tick() - t0 < 0.6 do
        if getCarriedEggTool() then ok = true break end
        doCarryRemote(ReturnState.uid, ReturnState.slotKey)
        task.wait(0.05)
    end

    if not ok then
        doCarryRemote(ReturnState.uid, ReturnState.slotKey)
        task.wait(0.1)
        ok = getCarriedEggTool() ~= nil
    end

    return ok
end

local function getFieldEggs()
    if not EggState or not EggState.ReadFieldEggs then return {} end
    local ok, snap = pcall(EggState.ReadFieldEggs)
    if not ok or not snap or not snap.Records then return {} end
    local out = {}
    for _, r in ipairs(snap.Records) do
        if r.State == "Slot" and r.BoundsCFrame then
            out[#out+1] = { record = r, info = getEggInfo(r) }
        end
    end
    table.sort(out, function(a, b) return a.info.rank > b.info.rank end)
    return out
end

local function passesFilter(entry)
    if HUB.minRarityRank > 0 and entry.info.rank < HUB.minRarityRank then return false end
    if HUB.biomeFilter ~= "Todos" then
        local id = tostring(entry.record.AreaId or "")
        if id:lower() ~= HUB.biomeFilter:lower() then return false end
    end
    return true
end

local function findRecord(uid)
    if not uid or not EggState or not EggState.ReadFieldEggs then return nil end
    local ok, snap = pcall(EggState.ReadFieldEggs)
    if not ok or not snap or not snap.Records then return nil end
    for _, r in ipairs(snap.Records) do
        if r.Uid == uid and r.State == "Slot" then return r end
    end
    return nil
end

local function stealEgg(record)
    if not record or not record.BoundsCFrame then return false end
    local uid = record.Uid
    if not uid then return false end

    print("[DARK HUB] Voando pro ovo (", HUB.flyMode, "):", getEggInfo(record).name)
    Fly.go(record.BoundsCFrame.Position + Vector3.new(0, 2, 0), 15)

    local _, hrp = getChar()
    if not hrp then return false end

    local slotKey = nil
    if AreaEggSlotId and AreaEggSlotId.LooksLikeFirstAreaUid
       and AreaEggSlotId.LooksLikeFirstAreaUid(uid) then
        slotKey = AreaEggSlotId.SlotKey(record.AreaId, record.NestId)
    end
    if EggWorld and EggWorld.AskFieldEggCarry then
        pcall(function() EggWorld.AskFieldEggCarry:InvokeServer({ Uid=uid, FirstAreaSlotKey=slotKey }) end)
    end
    pcall(function()
        if EggState.CarryFieldEgg then EggState.CarryFieldEgg(uid, slotKey) end
    end)

    local prompt = nil
    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("ProximityPrompt") and d.Name == "CarryAreaEgg" and d.Enabled then
            local p = d.Parent
            if p and p:IsA("Attachment") then p = p.Parent end
            if p and p:IsA("BasePart") and (p.Position - hrp.Position).Magnitude < 15 then
                prompt = d break
            end
        end
    end
    if prompt then
        prompt.HoldDuration = 0
        pcall(function() fireproximityprompt(prompt) end)
    end

    local t0 = tick()
    while tick() - t0 < 2 and not HUB.dead do
        if isCarryingEgg() then return true end
        pcall(function()
            if EggState.CarryFieldEgg then EggState.CarryFieldEgg(uid, slotKey) end
        end)
        if prompt then pcall(function() fireproximityprompt(prompt) end) end
        task.wait(0.1)
    end
    return isCarryingEgg()
end

local function getPlotPos()
    local ok, plot = pcall(function()
        return PlotState and PlotState.ResolvePlot and PlotState.ResolvePlot()
    end)
    if ok and plot then
        local cp = plot.CenterPoint
        local pt = cp and (typeof(cp) == "Vector3" and cp or cp.Position)
        if pt then return Vector3.new(pt.X, math.max(pt.Y, 70.4), pt.Z) end
    end
    return Vector3.new(464.7, 70.4, -364.0)
end

local function plantAllEggs()
    if not EggState or not EggState.PlantEgg or not EggToolDisplay then return 0 end
    local ids = {}
    for _, container in ipairs({LP.Character, LP:FindFirstChild("Backpack")}) do
        if container then
            for _, t in ipairs(container:GetChildren()) do
                if t:IsA("Tool") and EggToolDisplay.IsEggTool and EggToolDisplay.IsEggTool(t) then
                    local uid = EggToolDisplay.GetToolUid and EggToolDisplay.GetToolUid(t)
                    if uid then table.insert(ids, uid) end
                end
            end
        end
    end
    local count = 0
    for _, uid in ipairs(ids) do
        local ok, res = pcall(function()
            return EggState.PlantEgg(uid, CFrame.new(math.random(-6,6), 0, math.random(-6,6)))
        end)
        if ok and res then count = count + 1 end
        task.wait(0.05)
    end
    return count
end

local function returnHome()
    if not getCarriedEggTool() and not isCarryingEgg() then return false end
    captureReturnContext()
    print("[DARK HUB] Voltando | uid:", tostring(ReturnState.uid), "| modo:", HUB.flyMode)

    local targetPos = getPlotPos() + Vector3.new(0, 2, 0)
    local _, hrp = getChar()
    if not hrp then return false end

    local startPos = hrp.Position
    local delta = targetPos - startPos
    local total = delta.Magnitude

    if total <= HUB.finalFlyDist then
        Fly.go(targetPos, 10)
        if HUB.dropPickEnabled then dropPickIsland() end
        plantAllEggs()
        ReturnState.uid = nil ReturnState.slotKey = nil
        return true
    end

    local dir = delta.Unit
    local tpDist = total - HUB.finalFlyDist
    local steps = math.max(1, math.ceil(tpDist / HUB.stepSize))
    local realStep = tpDist / steps

    print(string.format("[DARK HUB] Retorno: %d ilhas | drop/pick em cada | ultimos %d studs voando",
        steps, HUB.finalFlyDist))

    for i = 1, steps do
        if HUB.dead or not HUB.running then return false end
        local _, r = getChar()
        if not r then return false end

        local nextPos = startPos + dir * (realStep * i)
        pcall(function()
            r.CFrame = CFrame.new(nextPos)
            r.AssemblyLinearVelocity = Vector3.zero
            r.AssemblyAngularVelocity = Vector3.zero
        end)

        SetSpoofWalkSpeed(HUB.flySpeed)
        if BypassState.integrityState then
            SpoofState(BypassState.integrityState, HUB.flySpeed * HEADROOM)
        end
        SyncHoverPad()

        if HUB.dropPickEnabled then
            dropPickIsland()
        end
        task.wait(HUB.stepDelay)
    end

    Fly.go(targetPos, 10)

    if HUB.dropPickEnabled then
        dropPickIsland()
        task.wait(0.1)
    end

    plantAllEggs()
    ReturnState.uid = nil ReturnState.slotKey = nil
    print("[DARK HUB] Ciclo completo")
    return true
end

local function runCycle()
    if HUB.busy or HUB.dead then return end
    HUB.busy = true

    if isCarryingEgg() then
        returnHome()
        HUB.busy = false
        return
    end

    if HUB.activeTarget and HUB.activeTarget.uid then
        local rec = findRecord(HUB.activeTarget.uid)
        if rec then
            if stealEgg(rec) then
                HUB.activeTarget = nil
                returnHome()
            end
            HUB.busy = false
            return
        else HUB.activeTarget = nil end
    end

    local eggs = getFieldEggs()
    if #eggs == 0 then HUB.busy = false return end

    local filtered = {}
    for _, e in ipairs(eggs) do
        if passesFilter(e) then filtered[#filtered+1] = e end
    end
    if #filtered == 0 then filtered = eggs end

    local target = filtered[1]
    if HUB.selectedUid then
        for _, e in ipairs(filtered) do
            if e.record.Uid == HUB.selectedUid then target = e break end
        end
    end

    HUB.activeTarget = { uid=target.record.Uid, name=target.info.name }
    if stealEgg(target.record) then
        HUB.activeTarget = nil
        returnHome()
    end
    HUB.busy = false
end

local Esp = { container=nil, parts={} }
local function getEspFolder()
    if Esp.container and Esp.container.Parent then return Esp.container end
    local p = nil
    pcall(function() p = gethui and gethui() or game:GetService("CoreGui") end)
    if not p then p = PGui end
    Esp.container = Instance.new("Folder")
    Esp.container.Name = "DarkHubEsp"
    pcall(function() Esp.container.Parent = p end)
    return Esp.container
end
local function updateEsp()
    if not HUB.espEnabled then
        for k, v in pairs(Esp.parts) do if v.part then v.part:Destroy() end Esp.parts[k] = nil end
        return
    end
    local folder = getEspFolder()
    local eggs = getFieldEggs()
    local seen = {}
    for _, e in ipairs(eggs) do
        local uid = e.record.Uid
        if uid then
            seen[uid] = true
            local data = Esp.parts[uid]
            if not data or not data.part.Parent then
                local part = Instance.new("Part")
                part.Size = Vector3.new(0.2,0.2,0.2) part.Transparency = 1
                part.Anchored = true part.CanCollide = false
                part.CanQuery = false part.CanTouch = false
                part.Parent = folder
                local bill = Instance.new("BillboardGui")
                bill.Adornee = part bill.Size = UDim2.fromOffset(140, 50)
                bill.StudsOffset = Vector3.new(0, 4, 0)
                bill.AlwaysOnTop = true bill.MaxDistance = 900 bill.Parent = part
                local icon = Instance.new("ImageLabel")
                icon.Size = UDim2.fromOffset(30,30)
                icon.Position = UDim2.new(0.5, -15, 0, 0)
                icon.BackgroundTransparency = 1
                icon.ScaleType = Enum.ScaleType.Fit icon.Parent = bill
                local label = Instance.new("TextLabel")
                label.Position = UDim2.new(0, 0, 0, 32)
                label.Size = UDim2.new(1, 0, 0, 16)
                label.BackgroundTransparency = 1
                label.Font = Enum.Font.GothamBold label.TextSize = 12
                label.TextStrokeTransparency = 0.4 label.Parent = bill
                Esp.parts[uid] = { part=part, icon=icon, label=label }
                data = Esp.parts[uid]
            end
            data.part.CFrame = CFrame.new(e.record.BoundsCFrame.Position)
            data.label.Text = string.format("[%s] %s", e.info.rarity, e.info.name)
            data.label.TextColor3 = e.info.color or Color3.fromRGB(255,200,50)
            if e.info.icon then data.icon.Image = e.info.icon data.icon.Visible = true
            else data.icon.Visible = false end
        end
    end
    for k, v in pairs(Esp.parts) do
        if not seen[k] then v.part:Destroy() Esp.parts[k] = nil end
    end
end
task.spawn(function()
    while true do pcall(updateEsp) task.wait(0.5) end
end)

local Webhook = { url="", enabled=false }
local reportedEggs = {}
local function sendWebhook(eggInfo)
    if not Webhook.enabled or Webhook.url == "" or not eggInfo then return end
    local key = eggInfo.name .. "|" .. eggInfo.rarity
    if reportedEggs[key] and tick() - reportedEggs[key] < 60 then return end
    reportedEggs[key] = tick()
    local payload = { content = string.format("Egg **%s** [%s] pego!", eggInfo.name, eggInfo.rarity) }
    pcall(function()
        local HttpService = game:GetService("HttpService")
        local req = http_request or request or (syn and syn.request)
        if req then
            req({
                Url = Webhook.url, Method = "POST",
                Headers = { ["Content-Type"] = "application/json" },
                Body = HttpService:JSONEncode(payload)
            })
        end
    end)
end

local Bloom = { autoNet=false, autoCatch=false, autoTrade=false, autoCraft=false }
local function bloomSnapshot()
    local ok, save = pcall(function() return (SaveModule.Peek or SaveModule.Get)() end)
    if not ok or not save then return {} end
    local bf = save.Butterflies or {}
    local endsAt = workspace:GetAttribute("ButterflyBloomEndsAt") or 0
    local now = workspace:GetServerTimeNow()
    return {
        active = (endsAt - now) > 0,
        left = math.max(0, endsAt - now),
        net = bf.NetClaimed == true,
        essence = tonumber(bf.Essence) or 0,
        caught = tonumber(bf.TotalCaught) or 0,
    }
end
local function bloomClaimNet()
    pcall(function() Remotes.Butterflies.AskClaimNet:InvokeServer() end)
end
local function bloomSwing()
    local swarm = Workspace:FindFirstChild("ButterflyBloomSwarm")
    if not swarm then return end
    local _, hrp = getChar()
    if not hrp then return end
    local best, bestDist = nil, 20
    for _, m in ipairs(swarm:GetChildren()) do
        if m:IsA("Model") and m.PrimaryPart then
            local d = (m.PrimaryPart.Position - hrp.Position).Magnitude
            if d < bestDist then best, bestDist = m, d end
        end
    end
    if best then pcall(function() Remotes.Butterflies.AskSwing:InvokeServer(best.Name) end) end
end

local Scramble = { autoBuy=false, autoTrade=false, autoReroll=false, autoBoss=false }
local function scrambleSnapshot()
    if not Remotes or not Remotes.Scramble then return nil end
    local ok, snap = pcall(function() return Remotes.Scramble.Request:InvokeServer("Snapshot") end)
    if ok and type(snap) == "table" then return snap end
end
local function scrambleBuyAll()
    local snap = scrambleSnapshot()
    if not snap or not snap.Shop then return end
    local samples = (snap.State and snap.State.Samples) or 0
    for _, item in pairs(snap.Shop) do
        if (item.Price or 0) <= samples then
            pcall(function()
                Remotes.Scramble.Request:InvokeServer("Shop", item.Id, {
                    Quote = item.Quote, Sequence = snap.State.ShopSequence,
                })
            end)
            task.wait(0.3)
        end
    end
end
local function scrambleBossHit()
    local arena = Workspace:FindFirstChild("ScrambleArena")
    if not arena then return end
    local _, hrp = getChar()
    if not hrp then return end
    local target = nil
    for _, name in ipairs({"Mech","Ball","Croc","Human","DrScramble","Pilot"}) do
        local m = arena:FindFirstChild(name)
        if m then
            local p = m.PrimaryPart or m:FindFirstChild("Hitbox", true)
            if p and (p.Position - hrp.Position).Magnitude < 60 then target = p break end
        end
    end
    if target then
        local tool = LP.Character and LP.Character:FindFirstChildWhichIsA("Tool")
        if tool then pcall(function() tool:Activate() end) end
        pcall(function()
            if Remotes.BatSwing and Remotes.BatSwing.Trigger then
                Remotes.BatSwing.Trigger:FireServer()
            end
        end)
    end
end
task.spawn(function()
    while true do
        if Scramble.autoBuy then pcall(scrambleBuyAll) end
        task.wait(3)
    end
end)
task.spawn(function()
    while true do
        if Scramble.autoTrade then
            pcall(function() Remotes.ScrambleTradeIn.AskTradeIn:InvokeServer() end)
        end
        task.wait(2)
    end
end)
task.spawn(function()
    while true do
        if Scramble.autoReroll then
            pcall(function() Remotes.ScrambleTradeIn.AskRefresh:InvokeServer() end)
        end
        task.wait(3)
    end
end)
task.spawn(function()
    while true do
        if Scramble.autoBoss then
            if LP:GetAttribute("InScrambleArena") == true then
                scrambleBossHit()
                task.wait(0.15)
            else
                pcall(function()
                    local portal = Workspace:FindFirstChild("ScrambleArenaPortal")

                        local _, hrp = getChar()
                        if hrp then hrp.CFrame = CFrame.new(portal.Position + Vector3.new(0, 3, 0)) end
                    end
                end)
                task.wait(2)
            end
        else task.wait(1) end
    end
end)
